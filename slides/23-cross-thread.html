Why MessagePort is slow

// 1. structured clone: a deep copy  2. queue it across the boundary
// 3. deserialize into new objects   4. the GC collects both copies

port.postMessage(buf, [buf.buffer])
// transfer: no copy, but the sender's buffer is detached,
// a new wrapper is created, and you can only move, never share

API

import { Writer, Reader } from '@nxtedition/shared'

// writer thread
const w = new Writer(8 * 1024 * 1024)
new Worker('./reader.js', { workerData: w.handle })
w.writeSync(buf.length, (d, b) => d.byteOffset + b.copy(d.buffer, d.byteOffset), buf)

// reader thread: zero-copy views into shared memory
const r = new Reader(workerData)
r.readSome(({ buffer, byteOffset, byteLength }) => {
  process(buffer, byteOffset, byteLength)
})
