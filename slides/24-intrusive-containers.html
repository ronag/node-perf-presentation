Standard vs intrusive O(1) removal

Standard: wrapper nodes or scans
set.add(item)       // internal entry → GC
set.delete(item)

list.add(item)      // wrapper node → GC
list.remove(item)

arr.push(item)
arr.splice(arr.indexOf(item), 1)   // O(n)
Intrusive: zero allocations
function add (arr, item) {
  item[kIndex] = arr.length
  arr.push(item)
}
function remove (arr, item) {   // O(1) swap-remove
  const last = arr.pop()
  if (last !== item) {
    arr[item[kIndex]] = last
    last[kIndex] = item[kIndex]
  }
  item[kIndex] = -1
}
