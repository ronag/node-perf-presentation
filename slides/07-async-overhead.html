The async return pattern

// SOMETIMES async, but the caller always pays
async function resolve (key) {
  const cached = cache.get(key)
  if (cached) return cached          // already have it, but…
  return await fetchRemote(key)
}

const value = await resolve(key)     // a Promise even on a hit

The async return pattern: solution

function resolve (key) {
  const cached = cache.get(key)
  return cached !== undefined
    ? { async: false, value: cached }
    : { async: true, value: fetchRemote(key) }
}

const r = resolve(key)
const value = r.async ? await r.value : r.value

The result object is usually removed by escape analysis. If it escapes, reuse one.

Async return pattern: real world

function processMessage (msg) {
  const r = resolve(msg.key)
  if (r.async) return r.value.then((v) => handle(msg, v))  // rare
  handle(msg, r.value)                                      // hot path
}
